cmake_minimum_required(VERSION 3.20)
if(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
    project(IFDDAMishchenko LANGUAGES CXX)
endif()
option(IFDDA_MISHCHENKO_BUILD_STANDALONE "Build all three standalone reference executables" ON)
option(IFDDA_MISHCHENKO_ENABLE_CUDA "Build OpenMP + CUDA reference" ON)
if(NOT IFDDA_MISHCHENKO_BUILD_STANDALONE AND NOT IFDDA_MISHCHENKO_ENABLE_CUDA)
    message(FATAL_ERROR "The IF-DDA reference library requires the OpenMP + CUDA backend.")
endif()
if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
set(mish_dir "${CMAKE_CURRENT_SOURCE_DIR}")
set(nasa_dir "${mish_dir}/NASA")
set(work_dir "${CMAKE_CURRENT_BINARY_DIR}/generated")
set(core "${work_dir}/mish_core.f" "${work_dir}/lpd.f" "${work_dir}/mish_qp.f")
set(wrapper "${mish_dir}/mishchenko_reference.f90")
set(fortran_flags -std=legacy -fallow-argument-mismatch -ffixed-line-length-none -ffree-line-length-none -cpp)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
foreach(f ampld.lp.f ampld.par.f lpd.f amplq.lp.f amplq.par.f lpq.f)
    if(NOT EXISTS "${nasa_dir}/${f}")
        message(FATAL_ERROR "Obtain the original NASA fixed-orientation source: ${nasa_dir}/${f}")
    endif()
endforeach()
add_custom_command(OUTPUT ${core}
    COMMAND Python3::Interpreter "${mish_dir}/prepare_core.py" --output-dir "${work_dir}"
    DEPENDS "${mish_dir}/prepare_core.py" "${mish_dir}/tmatrix_openmp.py" "${mish_dir}/quad_intrinsics.py"
        "${nasa_dir}/ampld.lp.f" "${nasa_dir}/ampld.par.f" "${nasa_dir}/lpd.f" "${nasa_dir}/amplq.lp.f" "${nasa_dir}/amplq.par.f" "${nasa_dir}/lpq.f"
    VERBATIM)
add_custom_target(ifdda_mishchenko_generated DEPENDS ${core})

if(WIN32)
    # The Fortran ABI is always built by x64 GNU Fortran. The GPU kernel
    # is compiled separately by nvcc/MSVC, independent of the C++ toolchain.
    if(MINGW AND CMAKE_Fortran_COMPILER AND NOT IFDDA_MISHCHENKO_GFORTRAN)
        set(IFDDA_MISHCHENKO_GFORTRAN "${CMAKE_Fortran_COMPILER}" CACHE FILEPATH "GNU Fortran compiler")
    endif()
    find_program(IFDDA_MISHCHENKO_GFORTRAN NAMES gfortran
        HINTS "$ENV{IFDDA_MISHCHENKO_MINGW_BIN}" "F:/mingw64_11.2/bin" REQUIRED)
    execute_process(COMMAND "${IFDDA_MISHCHENKO_GFORTRAN}" -dumpmachine
        OUTPUT_VARIABLE fortran_target OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE status)
    if(NOT status EQUAL 0 OR NOT fortran_target MATCHES "^x86_64.*mingw")
        message(FATAL_ERROR "Select a 64-bit MinGW-w64 GNU Fortran compiler.")
    endif()
    get_filename_component(gnu_bin "${IFDDA_MISHCHENKO_GFORTRAN}" DIRECTORY)
    find_program(MISH_DLLTOOL NAMES dlltool HINTS "${gnu_bin}" NO_DEFAULT_PATH REQUIRED)
    set(out "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>")
    if(IFDDA_MISHCHENKO_ENABLE_CUDA)
        set(kernel_dll "${out}/ifdda_mishchenko_cuda_kernel.dll")
        set(kernel_import "${out}/libifdda_mishchenko_cuda_kernel.dll.a")
        set(cuda_runtime "${out}/cudart64_12.dll")
        add_custom_command(OUTPUT "${kernel_dll}" "${kernel_import}" "${cuda_runtime}"
            COMMAND "${CMAKE_COMMAND}" -E make_directory "${out}"
            COMMAND cmd /c call "${mish_dir}/build_cuda.bat" "${out}"
            COMMAND "${MISH_DLLTOOL}" -d "${mish_dir}/mish_cuda.def" -l "${kernel_import}"
            DEPENDS "${mish_dir}/build_cuda.bat" "${mish_dir}/mish_cuda.cu" "${mish_dir}/mish_cuda.def"
            VERBATIM)
        add_custom_target(ifdda_mishchenko_kernel DEPENDS "${kernel_dll}" "${kernel_import}" "${cuda_runtime}")
    endif()
else()
    enable_language(Fortran)
    if(NOT CMAKE_Fortran_COMPILER_ID STREQUAL "GNU")
        message(FATAL_ERROR "Select GNU Fortran for the NASA legacy sources.")
    endif()
    find_package(OpenMP REQUIRED COMPONENTS Fortran)
    if(IFDDA_MISHCHENKO_ENABLE_CUDA)
        if(NOT CMAKE_CUDA_COMPILER AND "$ENV{CUDACXX}" STREQUAL "")
            find_program(MISH_NVCC NAMES nvcc HINTS "${CUDAToolkit_ROOT}/bin" /usr/local/cuda/bin REQUIRED)
            set(CMAKE_CUDA_COMPILER "${MISH_NVCC}" CACHE FILEPATH "CUDA compiler" FORCE)
        endif()
        set(CMAKE_CUDA_ARCHITECTURES 70 CACHE STRING "CUDA GPU architectures")
        enable_language(CUDA)
        find_package(CUDAToolkit REQUIRED)
    endif()
endif()

function(mish_variant suffix backend output_name)
    set(target ifdda_mishchenko_${suffix})
    set(lib ${target}_link)
    if(backend EQUAL 2)
        set(target ifdda_mishchenko_shared)
        set(lib ifdda_mishchenko_link)
    endif()
    set(flags)
    if(backend GREATER 0)
        list(APPEND flags -fopenmp -fmax-stack-var-size=32768)
    endif()
    if(backend EQUAL 2)
        list(APPEND flags -DMISH_CUDA)
    endif()
    if(WIN32)
        set(dll "${out}/${output_name}.dll")
        set(gnu_import "${out}/lib${output_name}.dll.a")
        set(def "${work_dir}/${output_name}.def")
        file(GENERATE OUTPUT "${def}" CONTENT "LIBRARY ${output_name}.dll\nEXPORTS\n    ifdda_mishchenko_v1\n    ifdda_mishchenko_v2\n    ifdda_mishchenko_backend\n")
        set(link_deps)
        if(backend EQUAL 2)
            list(APPEND link_deps "${kernel_import}")
        endif()
        set(msvc_commands)
        set(outputs "${dll}" "${gnu_import}")
        if(MSVC)
            set(msvc_import "${out}/${output_name}.lib")
            list(APPEND outputs "${msvc_import}")
            set(msvc_commands COMMAND "${CMAKE_AR}" /nologo /machine:x64 "/def:${def}" "/out:${msvc_import}")
        endif()
        add_custom_command(OUTPUT ${outputs}
            COMMAND "${CMAKE_COMMAND}" -E make_directory "${out}/${suffix}_modules"
            COMMAND "${IFDDA_MISHCHENKO_GFORTRAN}" -O3 ${fortran_flags} ${flags}
                "-I${nasa_dir}" "-J${out}/${suffix}_modules" -shared -static -static-libgfortran -static-libgcc
                "-Wl,--out-implib,${gnu_import}" -o "${dll}" ${core} "${wrapper}" "${def}" ${link_deps}
            ${msvc_commands}
            DEPENDS ${core} "${wrapper}" "${def}" "${nasa_dir}/ampld.par.f" ${link_deps}
            VERBATIM)
        add_custom_target(${target} ALL DEPENDS ${outputs})
        add_dependencies(${target} ifdda_mishchenko_generated)
        if(backend EQUAL 2)
            add_dependencies(${target} ifdda_mishchenko_kernel)
        endif()
        add_library(${lib} SHARED IMPORTED GLOBAL)
        foreach(config IN LISTS CMAKE_CONFIGURATION_TYPES CMAKE_BUILD_TYPE)
            string(TOUPPER "${config}" config_upper)
            if(MSVC)
                set(implib "${CMAKE_CURRENT_BINARY_DIR}/${config}/${output_name}.lib")
            else()
                set(implib "${CMAKE_CURRENT_BINARY_DIR}/${config}/lib${output_name}.dll.a")
            endif()
            set_target_properties(${lib} PROPERTIES
                "IMPORTED_LOCATION_${config_upper}" "${CMAKE_CURRENT_BINARY_DIR}/${config}/${output_name}.dll"
                "IMPORTED_IMPLIB_${config_upper}" "${implib}")
        endforeach()
        add_dependencies(${lib} ${target})
        if(backend EQUAL 2)
            set_property(GLOBAL PROPERTY IFDDA_MISHCHENKO_WINDOWS_RUNTIME "${dll};${kernel_dll};${cuda_runtime}")
        endif()
    else()
        add_library(${target} SHARED ${core} "${wrapper}")
        add_dependencies(${target} ifdda_mishchenko_generated)
        file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/modules_${suffix}")
        set_target_properties(${target} PROPERTIES OUTPUT_NAME "${output_name}"
            Fortran_MODULE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/modules_${suffix}"
            Fortran_PREPROCESS ON)
        target_include_directories(${target} PRIVATE "${nasa_dir}")
        target_compile_options(${target} PRIVATE $<$<COMPILE_LANGUAGE:Fortran>:${fortran_flags}>)
        if(backend GREATER 0)
            target_compile_options(${target} PRIVATE $<$<COMPILE_LANGUAGE:Fortran>:-fmax-stack-var-size=32768>)
            target_link_libraries(${target} PRIVATE OpenMP::OpenMP_Fortran)
        endif()
        if(backend EQUAL 2)
            target_sources(${target} PRIVATE "${mish_dir}/mish_cuda.cu")
            target_compile_definitions(${target} PRIVATE MISH_CUDA)
            target_compile_options(${target} PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
            target_link_libraries(${target} PRIVATE CUDA::cudart)
            set_target_properties(${target} PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES
                CUDA_RUNTIME_LIBRARY Shared LINKER_LANGUAGE CXX)
        endif()
        add_library(${lib} ALIAS ${target})
    endif()

    if(IFDDA_MISHCHENKO_BUILD_STANDALONE)
        set(exe ifdda_mishchenko_sections_${suffix})
        add_executable(${exe} "${mish_dir}/sections_main.cpp")
        target_compile_features(${exe} PRIVATE cxx_std_17)
        target_link_libraries(${exe} PRIVATE ${lib})
        if(WIN32)
            set_target_properties(${exe} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${out}")
            if(MINGW)
                target_link_options(${exe} PRIVATE -static)
            endif()
        else()
            set_target_properties(${exe} PROPERTIES BUILD_RPATH_USE_ORIGIN ON)
        endif()
    endif()
endfunction()
if(IFDDA_MISHCHENKO_BUILD_STANDALONE)
    mish_variant(cpu 0 ifdda_mishchenko_cpu)
    mish_variant(openmp 1 ifdda_mishchenko_openmp)
endif()
if(IFDDA_MISHCHENKO_ENABLE_CUDA)
    mish_variant(cuda 2 ifdda_mishchenko)
endif()

# Attach the fast reference to any IF-DDA target. On Windows the existing
# C ABI loader resolves the library beside the executable.
function(ifdda_mishchenko_attach target)
    if(WIN32)
        get_property(runtime GLOBAL PROPERTY IFDDA_MISHCHENKO_WINDOWS_RUNTIME)
        get_target_property(_destination ${target} RUNTIME_OUTPUT_DIRECTORY)
        if(NOT _destination)
            get_target_property(_destination ${target} BINARY_DIR)
        endif()
        string(MD5 _key "${_destination}")
        set(_deploy ifdda_mishchenko_runtime_${_key})
        if(NOT TARGET ${_deploy})
            add_custom_target(${_deploy}
                COMMAND "${CMAKE_COMMAND}" -E make_directory "$<TARGET_FILE_DIR:${target}>"
                COMMAND "${CMAKE_COMMAND}" -E copy_if_different ${runtime} "$<TARGET_FILE_DIR:${target}>"
                VERBATIM)
            add_dependencies(${_deploy} ifdda_mishchenko_shared)
        endif()
        add_dependencies(${target} ${_deploy})
    else()
        target_link_libraries(${target} PUBLIC ifdda_mishchenko_link)
        target_compile_definitions(${target} PRIVATE IFDDA_LINK_MISHCHENKO)
    endif()
endfunction()
